\section{Full Mathematical Derivation}

\subsection{Total Amplitude}
\begin{equation}
A_{ij} = \sum_{k} c_k F_{ijk}
\end{equation}

\subsection{Signal Probability Density (per event)}
\begin{equation}
S_i = \sum_{j} |A_{ij}|^2 = \sum_{j} \left|\sum_{k} c_k F_{ijk}\right|^2 = \sum_{j} \sum_{k} \sum_{k'} c_k c_{k'}^* F_{ijk} F_{ijk'}^*
\end{equation}

\subsection{Normalization Factors}

\textbf{Signal normalization:}
\begin{equation}
N_s = \sum_{i'} \omega_{i'} S_{i'} = \sum_{i'} \omega_{i'} \sum_{j} \sum_{k} \sum_{k'} c_k c_{k'}^* F_{i'jk} F_{i'jk'}^* = \sum_{k} \sum_{k'} c_k c_{k'}^* M_{kk'}
\end{equation}
where the overlap matrix $M_{kk'}$ is:
\begin{equation}
M_{kk'} = \sum_{i'} \sum_{j} \omega_{i'} F_{i'jk} F_{i'jk'}^*
\end{equation}

\textbf{Background normalization:}
\begin{equation}
N_b = \sum_{i'} \omega_{i'} B_{i'}
\end{equation}

\subsection{Total Probability Density}
\begin{equation}
P_i = \frac{S_i}{N_s} \cdot \text{purity} + \frac{B_i}{N_b} \cdot (1 - \text{purity})
\end{equation}

\subsection{Negative Log-Likelihood}
\begin{equation}
-\ln L = -\sum_i w_i \ln P_i
\end{equation}

\subsection{Gradients}

\textbf{Gradient of $S_i$ w.r.t. $c_k^*$:}
\begin{equation}
\frac{\partial S_i}{\partial c_k^*} = \sum_{j} F_{ijk}^* A_{ij} = \sum_{j} F_{ijk}^* \sum_{k'} c_{k'} F_{ijk'}
\end{equation}

\textbf{Gradient of $N_s$ w.r.t. $c_k^*$:}
\begin{equation}
\frac{\partial N_s}{\partial c_k^*} = \sum_{k'} c_{k'} M_{k'k}
\end{equation}

\textbf{Gradient of $P_i$ w.r.t. $c_k^*$:}
\begin{equation}
\frac{\partial P_i}{\partial c_k^*} = p \cdot \frac{\partial}{\partial c_k^*}\left(\frac{S_i}{N_s}\right) = p \cdot \frac{\frac{\partial S_i}{\partial c_k^*} N_s - S_i \frac{\partial N_s}{\partial c_k^*}}{N_s^2} = \frac{p}{N_s}\left(\frac{\partial S_i}{\partial c_k^*} - \frac{S_i}{N_s} \frac{\partial N_s}{\partial c_k^*}\right)
\end{equation}
where $p = \text{purity}$.

\textbf{Gradient of $-\ln L$ w.r.t. $c_k^*$:}
\begin{equation}
\frac{\partial(-\ln L)}{\partial c_k^*} = -\sum_i \frac{w_i}{P_i} \frac{\partial P_i}{\partial c_k^*} = -\frac{p}{N_s} \sum_i \frac{w_i}{P_i} \left(\frac{\partial S_i}{\partial c_k^*} - \frac{S_i}{N_s} \frac{\partial N_s}{\partial c_k^*}\right)
\end{equation}
Expanding to make the $N_s$ contribution explicit:
\begin{equation}
\frac{\partial(-\ln L)}{\partial c_k^*} = \underbrace{-\frac{p}{N_s} \sum_i \frac{w_i}{P_i} \frac{\partial S_i}{\partial c_k^*}}_{\text{data term}} + \underbrace{\frac{p}{N_s^2} \frac{\partial N_s}{\partial c_k^*} \sum_i \frac{w_i S_i}{P_i}}_{\text{normalization correction from } N_s}
\end{equation}
The second term is critical: it accounts for the fact that changing $c_k^*$ changes $N_s$, which rescales the entire signal PDF. Without this term, the fit would not properly normalize the probability density.

\textbf{Using complex conjugate symmetry:}
\begin{equation}
\frac{\partial(-\ln L)}{\partial c_k} = \left(\frac{\partial(-\ln L)}{\partial c_k^*}\right)^*
\end{equation}
This halves the gradient computation cost.

\subsection{Key Observations}

\begin{enumerate}
\item \textbf{Pre-computing $M_{kk'}$} is the most expensive one-time step ($2 \times 10^{11}$ ops), but it's done only once and cached.

\item \textbf{Per iteration}, the dominant costs are:
\begin{itemize}
\item Computing $A_{ij}$: $2 \times 10^8$ ops
\item Computing $\partial S_i/\partial c_k^*$: $2 \times 10^8$ ops
\item Computing $\partial P_i/\partial c_k^*$ and $\partial(-\ln L)/\partial c_k^*$: $2 \times 10^8$ ops
\end{itemize}

\item Using the $M_{kk'}$ matrix reduces the per-iteration normalization cost from $O(i' \cdot j \cdot k)$ to $O(k^2)$, a speedup factor of $\sim 10^5$.

\item \textbf{Memory bottleneck}: Storing $F_{i'jk}$ for MC requires $\sim$32 GB. If memory is constrained, $M_{kk'}$ can be computed on-the-fly in batches.

\item \textbf{Using complex conjugate symmetry}: $\partial(-\ln L)/\partial c_k = [\partial(-\ln L)/\partial c_k^*]^*$, which halves the gradient computation cost.
\end{enumerate}

\section{Optimized Computation Order}

The naive approach computes per-event, per-component intermediate quantities unnecessarily. The key optimization is to \textbf{restructure all computations as batched matrix multiplications (GEMM/GEMV)}, which enable BLAS/GPU acceleration and eliminate redundant storage.

Define the matrix $F^{(j)}$ of shape $(N, k)$ with elements $F^{(j)}_{nk} = F_{njk}$, where $N$ is the number of events.

\subsection{Gradient Reformulation}

The data term in the gradient can be rewritten as:
\begin{equation}
\sum_i \frac{w_i}{P_i} \frac{\partial S_i}{\partial c_k^*} = \sum_i \frac{w_i}{P_i} \sum_j F_{ijk}^* A_{ij} = \sum_j \sum_i G_{ij} F_{ijk}^*
\end{equation}
where $G_{ij} \equiv \frac{w_i}{P_i} A_{ij}$. This is a \textbf{batched GEMM}:
\begin{equation}
g^{\text{data}} = \sum_j \left(F_{\text{data}}^{(j)}\right)^\dagger G^{(j)}
\end{equation}
with shape $(k, i) \times (i, 1) \to (k, 1)$.

The normalization correction term is a simple scalar sum:
\begin{equation}
S_{\text{corr}} = \sum_i \frac{w_i S_i}{P_i}
\end{equation}

The full gradient is then assembled as an $O(k)$ vector operation:
\begin{equation}
\frac{\partial(-\ln L)}{\partial c_k^*} = -\frac{p}{N_s} \, g_k^{\text{data}} + \frac{p}{N_s^2} \, \frac{\partial N_s}{\partial c_k^*} \, S_{\text{corr}}
\end{equation}

\textbf{This eliminates the $O(i \cdot k)$ per-event-per-k loops} for $\partial P_i/\partial c_k^*$ and $\partial(-\ln L)/\partial c_k^*$, saving $2 \times 10^8$ operations per iteration.

\subsection{Chunked Memory Optimization}

For $M_{kk'}$ computation, process MC events in chunks of size $C$:
\begin{equation}
M = \sum_{\text{chunks}} \sum_j \left(F^{(j)}_{\text{chunk}}\right)^\dagger \text{diag}(\omega_{\text{chunk}}) F^{(j)}_{\text{chunk}}
\end{equation}
With $C = 10^5$: peak memory drops from 35 GB to $\sim$320 MB, with identical total FLOPs.

\subsection{Optimized Algorithm}

\begin{verbatim}
Pre-computation:
  M = 0
  for each MC chunk:
    for j in {0, 1}:
      T = diag(omega_chunk) @ F_chunk[j]    # O(C*k)
      M += F_chunk[j].T @ T                  # O(C*k^2)
  N_b = sum(omega * B)                       # O(i')

Per iteration:
  for j in {0, 1}:
    A[j] = F_data[j] @ c                     # O(i*k) GEMV
  S = sum(|A|^2, axis=j)                     # O(i*j)
  N_s = c.T @ M @ c                          # O(k^2)
  P = S/N_s * p + B/N_b * (1-p)             # O(i)
  nll = -sum(w * log(P))                     # O(i)
  dN_s = M @ c                               # O(k^2)
  G = (w/P)[:,None] * A                      # O(i*j)
  g_data = sum(F_data[j].T @ G[j] for j)    # O(i*j*k) GEMM
  S_corr = sum(w * S / P)                   # O(i)
  grad = -p/N_s * g_data + p/N_s^2 * dN_s * S_corr  # O(k)
  return nll, grad, conj(grad)              # O(k)
\end{verbatim}
